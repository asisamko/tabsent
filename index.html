TabSent

Stop websites from noticing that you switched to another window, tab or app. The page always thinks it is visible and focused.

A page that watches your focus

Switch away and it notices — then turn the bypass on and it stops. Live counter, timer and event log.

1. Bookmarklet easiest

  1. Show the bookmarks bar: ⌘⇧B (Mac) / CtrlShiftB (Windows).
  2. Drag this button onto the bookmarks bar:
🙈 Drag to bookmarks bar

Then click the bookmark on the page where you want it. It stays active until you reload the page. Works almost everywhere; a few pages start watching focus before you can click the bookmark – use the userscript there.

2. Userscript most reliable

  1. Install the Tampermonkey or Violentmonkey extension.
  2. Click the button and confirm the install:
Install userscript

Runs automatically on every page, before the page's own scripts, and updates itself. Switch it off in Tampermonkey with one click.

3. Chrome extension per-site switch

Install the latest release directly — no repository download needed.

Download extension ZIPlatest release · v1.1.2
  1. Unzip the download.
  2. Open chrome://extensions and turn on Developer mode (top right).
  3. Click Load unpacked and pick the unzipped folder (the one with manifest.json).

Pin it to the toolbar, then click its icon on any site and switch it on. Chrome asks for access to just that site (or to all sites, if you choose that). The page reloads once so the guard starts before its scripts. Tabs where it is on show an ON badge.

4. Console no install

Temporary: it lasts until you reload the page. Good for a quick try without installing anything.

  1. Click Copy script.
  2. On the page you want to protect, open the console: ⌥⌘J (Mac) / CtrlShiftJ (Windows).
  3. Paste and press Enter. If Chrome asks, type allow pasting first.

(() => {
  const FLAG = Symbol.for('tabSent');
  if (window[FLAG]) return;
  Object.defineProperty(window, FLAG, { value: true });

  // Read the real state before overriding it: a page that starts in a background tab
  // should not see the focus event when the user first opens that tab either.
  const startedHidden = document.hidden;
  const nativeHidden = Object.getOwnPropertyDescriptor(Document.prototype, 'hidden')?.get;

  const getter = (proto, prop, value) => {
    try {
      Object.defineProperty(proto, prop, { configurable: true, enumerable: true, get: () => value });
    } catch {
      /* not configurable on this page */
    }
  };
  getter(Document.prototype, 'hidden', false);
  getter(Document.prototype, 'webkitHidden', false);
  getter(Document.prototype, 'visibilityState', 'visible');
  getter(Document.prototype, 'webkitVisibilityState', 'visible');

  const realHasFocus = Document.prototype.hasFocus;
  try {
    Document.prototype.hasFocus = function hasFocus() {
      return true;
    };
  } catch {
    /* ignore */
  }

  const isPageLevel = (t) => t === window || t === document || t === document.documentElement || t === document.body;
  // Element blur/focusout is swallowed only when the whole window lost focus,
  // so moving between fields on the page keeps working.
  const windowLostFocus = () => {
    try {
      return !realHasFocus.call(document);
    } catch {
      return false;
    }
  };

  // After a real blur, the browser fires focus/focusin again when the user comes back.
  // Hide those too, including the focus restored to the element that was active before.
  let returning = startedHidden;
  let returnTimer = 0;
  const finishReturning = () => {
    clearTimeout(returnTimer);
    returnTimer = setTimeout(() => {
      returning = false;
    }, 500);
  };
  const clearReturning = () => {
    returning = false;
    clearTimeout(returnTimer);
  };

  const stop = (e) => {
    if (isPageLevel(e.target) || windowLostFocus()) {
      clearTimeout(returnTimer);
      returning = true;
      e.stopImmediatePropagation();
    }
  };
  for (const type of ['blur', 'focusout']) window.addEventListener(type, stop, true);

  const stopFocus = (e) => {
    if (!returning) return;
    finishReturning();
    e.stopImmediatePropagation();
  };
  for (const type of ['focus', 'focusin']) window.addEventListener(type, stopFocus, true);

  // A real click or key press ends the "just came back" window, so focusing a field still works.
  for (const type of ['pointerdown', 'mousedown', 'keydown', 'touchstart']) {
    window.addEventListener(type, clearReturning, true);
  }

  const stopVisibility = (e) => {
    try {
      // The native getter still knows the truth even though document.hidden is overridden.
      if (nativeHidden?.call(document)) {
        clearTimeout(returnTimer);
        returning = true;
      }
    } catch {
      /* ignore */
    }
    e.stopImmediatePropagation();
  };
  for (const type of ['visibilitychange', 'webkitvisibilitychange']) window.addEventListener(type, stopVisibility, true);

  const stopLeave = (e) => {
    if (!e.relatedTarget && isPageLevel(e.target)) e.stopImmediatePropagation();
  };
  for (const type of ['mouseleave', 'mouseout', 'pointerleave', 'pointerout']) window.addEventListener(type, stopLeave, true);

  // Inline handlers (window.onblur = …, document.onvisibilitychange = …): drop any already set
  // and ignore new ones, so even a late injection silences them.
  const muteHandler = (obj, prop) => {
    try {
      obj[prop] = null;
      Object.defineProperty(obj, prop, { configurable: true, get: () => null, set: () => {} });
    } catch {
      /* ignore */
    }
  };
  for (const prop of ['onblur', 'onfocusout', 'onfocus', 'onfocusin', 'onvisibilitychange', 'onmouseleave', 'onmouseout']) {
    muteHandler(window, prop);
    muteHandler(document, prop);
  }
  if (document.readyState !== 'loading') console.info('TabSent: focus changes are now hidden from this page.');
})();
